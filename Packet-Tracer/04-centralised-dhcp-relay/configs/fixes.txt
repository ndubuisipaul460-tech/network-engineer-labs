! Fixes found when reviewing the as-built lab (see README "Review findings")

! --- R4: remove two static routes whose next hop (120.10.10.1) is not on any
! --- R4 interface. They can never be used; the correct routes via 120.10.10.9 stay.
enable
configure terminal
hostname R4
no ip route 100.150.50.0 255.255.255.224 120.10.10.1
no ip route 100.150.50.32 255.255.255.224 120.10.10.1
end
copy running-config startup-config

! --- R3: no DHCP clients live on the R3-R5 transit link, so no relay is needed there.
enable
configure terminal
hostname R3
interface g0/2
 no ip helper-address 150.30.30.2
end
copy running-config startup-config

! --- R5: name it
enable
configure terminal
hostname R5-CORE
end
copy running-config startup-config
